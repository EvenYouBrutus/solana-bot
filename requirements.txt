# Smart Money Monitor (Python) dependencies.
#
# Pinned versions are not used: Solana RPC tooling evolves quickly and
# the monitor is read-only, so a loose upper bound is sufficient.
#
# Installation:
#   python -m venv .venv
#   source .venv/bin/activate   (or .venv\Scripts\activate on Windows)
#   pip install -r requirements.txt

# Solana JSON-RPC client (HTTP) for history replay + read-only lookups.
solana>=0.30,<1.0

# Async WebSocket client for the real-time logsSubscribe / signatureSubscribe stream.
websockets>=12.0,<14.0

# HTTP client used for DexScreener / Birdeye / Jupiter price lookups
# (the on-chain RPC returns no USD price/liquidity directly).
requests>=2.31,<3.0

# base58 is already a transitive dependency of `solana` but we pin it
# explicitly because we use it directly for short mint decoding.
base58>=2.1,<3.0

# python-dotenv so `.env` is loaded transparently without a wrapper.
python-dotenv>=1.0,<2.0
